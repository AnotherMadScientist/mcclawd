```rust
// crates/mcclawd-swarm/src/dag.rs

use crate::error::{SwarmError, Result};
use mcclawd_core::types::SubtaskId;
use petgraph::graph::{DiGraph, NodeIndex};
use petgraph::algo::is_cyclic_directed;
use petgraph::visit::Topo;
use serde::{Deserialize, Serialize};
use std::collections::HashMap;

/// A node in the task DAG.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct DagNode {
    pub subtask_id: SubtaskId,
    pub agent_role: String,
    pub prompt: String,
    pub input_keys: Vec<String>,
    pub output_key: String,
}

/// DAG of subtasks with dependency edges.
pub struct TaskDag {
    graph: DiGraph<DagNode, ()>,
    id_to_index: HashMap<String, NodeIndex>,
}

impl TaskDag {
    pub fn new() -> Self {
        Self {
            graph: DiGraph::new(),
            id_to_index: HashMap::new(),
        }
    }

    pub fn add_node(&mut self, node: DagNode) -> NodeIndex {
        let id = node.subtask_id.to_string();
        let idx = self.graph.add_node(node);
        self.id_to_index.insert(id, idx);
        idx
    }

    pub fn add_edge(&mut self, from: NodeIndex, to: NodeIndex) -> Result<()> {
        self.graph.add_edge(from, to, ());
        if is_cyclic_directed(&self.graph) {
            // Remove the edge that caused the cycle
            let edge = self.graph.find_edge(from, to).unwrap();
            self.graph.remove_edge(edge);
            return Err(SwarmError::CycleDetected);
        }
        Ok(())
    }

    /// Returns nodes grouped into parallel execution waves (topological layers).
    pub fn topological_waves(&self) -> Vec<Vec<&DagNode>> {
        if self.graph.node_count() == 0 {
            return vec![];
        }

        let mut in_degree: HashMap<NodeIndex, usize> = HashMap::new();
        for idx in self.graph.node_indices() {
            in_degree.insert(idx, 0);
        }
        for edge in self.graph.edge_indices() {
            let (_, target) = self.graph.edge_endpoints(edge).unwrap();
            *in_degree.get_mut(&target).unwrap() += 1;
        }

        let mut waves = Vec::new();
        let mut remaining = in_degree;

        loop {
            let wave: Vec<NodeIndex> = remaining
                .iter()
                .filter(|(_, &deg)| deg == 0)
                .map(|(&idx, _)| idx)
                .collect();

            if wave.is_empty() {
                break;
            }

            for &idx in &wave {
                remaining.remove(&idx);
                for neighbor in self.graph.neighbors(idx) {
                    if let Some(deg) = remaining.get_mut(&neighbor) {
                        *deg -= 1;
                    }
                }
            }

            let wave_nodes: Vec<&DagNode> = wave
                .iter()
                .map(|&idx| &self.graph[idx])
                .collect();
            waves.push(wave_nodes);
        }

        waves
    }

    pub fn node_count(&self) -> usize {
        self.graph.node_count()
    }

    pub fn get_node(&self, id: &str) -> Option<&DagNode> {
        self.id_to_index.get(id).map(|&idx| &self.graph[idx])
    }
}
```

```rust
// crates/mcclawd-swarm/src/shared_memory.rs

use dashmap::DashMap;
use serde::{de::DeserializeOwned, Serialize};
use std::collections::HashMap;
use std::sync::Arc;

/// Thread-safe shared memory for swarm workers.
/// Keys are strings; values are stored as JSON for type flexibility.
#[derive(Clone)]
pub struct SharedMemory {
    store: Arc<DashMap<String, serde_json::Value>>,
}

impl SharedMemory {
    pub fn new() -> Self {
        Self {
            store: Arc::new(DashMap::new()),
        }
    }

    /// Set a value in shared memory (serialized to JSON).
    pub fn set<T: Serialize>(&self, key: &str, value: T) {
        let json = serde_json::to_value(value).expect("serialize shared memory value");
        self.store.insert(key.to_string(), json);
    }

    /// Get a value from shared memory (deserialized from JSON).
    pub fn get<T: DeserializeOwned>(&self, key: &str) -> Option<T> {
        self.store
            .get(key)
            .and_then(|v| serde_json::from_value(v.value().clone()).ok())
    }

    /// Check if a key exists.
    pub fn contains(&self, key: &str) -> bool {
        self.store.contains_key(key)
    }

    /// List all keys.
    pub fn keys(&self) -> Vec<String> {
        self.store.iter().map(|r| r.key().clone()).collect()
    }

    /// Get a snapshot of all key-value pairs.
    pub fn snapshot(&self) -> HashMap<String, serde_json::Value> {
        self.store
            .iter()
            .map(|r| (r.key().clone(), r.value().clone()))
            .collect()
    }
}
```

```rust
/// LLM-powered task decomposition planner.
/// Uses a Rig agent with CreateSubtask/AddDependency/FinalizePlan tools
/// to break a user prompt into a TaskDag.
pub struct SwarmPlanner {
    pub state: PlannerState,
    model: Option<String>,
    api_key: String,
}

impl SwarmPlanner {
    pub fn new(model: Option<String>, api_key: String) -> Self {
        Self {
            state: PlannerState::new(),
            model,
            api_key,
        }
    }

    /// Decompose a user prompt into a TaskDag using an LLM agent.
    pub async fn plan(&self, task_prompt: &str) -> crate::Result<TaskDag> {
        // Reset state for new planning session
        *self.state.dag.lock().unwrap() = TaskDag::new();
        *self.state.finalized.lock().unwrap() = false;

        let model_name = self.model.as_deref().unwrap_or("claude-sonnet-4-20250514");

        // Build a Rig agent with planner tools
        let client = rig::providers::anthropic::ClientBuilder::new(&self.api_key).build();
        let agent = client
            .agent(model_name)
            .preamble(PLANNER_PREAMBLE)
            .tool(CreateSubtaskTool::new(self.state.clone()))
            .tool(AddDependencyTool::new(self.state.clone()))
            .tool(FinalizePlanTool::new(self.state.clone()))
            .max_turns(20)
            .build();

        // Run the agent — it will call tools to build the DAG
        let _response = agent.prompt(task_prompt).await
            .map_err(|e| crate::SwarmError::PlanningFailed(e.to_string()))?;

        // Extract the finalized DAG
        let dag = self.state.dag.lock().unwrap();
        if dag.node_count() == 0 {
            return Err(crate::SwarmError::PlanningFailed("Planner produced empty DAG".into()));
        }
        // Clone and return
        Ok(dag.clone())
    }
}

const PLANNER_PREAMBLE: &str = r#"You are a task planner for a swarm of AI agents.
Given a user prompt, decompose it into subtasks that can be executed by specialized agents.

Use these tools to build the execution plan:
1. create_subtask — add a subtask with an agent role, prompt, and output key
2. add_dependency — specify that one subtask depends on another
3. finalize_plan — call when the plan is complete

Available agent roles: researcher, coder, writer, reviewer, analyst.
Each subtask writes to a unique output_key in shared memory.
Dependent subtasks read from input_keys.
Maximize parallelism: only add dependencies where truly needed.
"#;
```

```rust
// crates/mcclawd-swarm/src/merger.rs

use serde::{Deserialize, Serialize};

/// Strategy for merging multiple subtask outputs into a final result.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub enum MergeStrategy {
    /// Concatenate all outputs with headers.
    Concatenate,
    /// Use the output of a specific final subtask.
    LastWriterWins { final_key: String },
}

impl Default for MergeStrategy {
    fn default() -> Self {
        Self::Concatenate
    }
}

/// Merges subtask outputs into a single final output.
pub struct OutputMerger {
    strategy: MergeStrategy,
}

impl OutputMerger {
    pub fn new(strategy: MergeStrategy) -> Self {
        Self { strategy }
    }

    pub fn merge(&self, outputs: &[(String, String)]) -> String {
        match &self.strategy {
            MergeStrategy::Concatenate => {
                outputs
                    .iter()
                    .map(|(key, val)| format!("## {}\n\n{}", key, val))
                    .collect::<Vec<_>>()
                    .join("\n\n")
            }
            MergeStrategy::LastWriterWins { final_key } => {
                outputs
                    .iter()
                    .find(|(k, _)| k == final_key)
                    .map(|(_, v)| v.clone())
                    .unwrap_or_else(|| {
                        // Fallback to concatenation
                        OutputMerger::new(MergeStrategy::Concatenate).merge(outputs)
                    })
            }
        }
    }
}
```

```rust
// crates/mcclawd-swarm/src/worker.rs

use crate::shared_memory::SharedMemory;
use crate::dag::DagNode;
use mcclawd_core::types::SubtaskId;
use serde::{Deserialize, Serialize};
use std::time::Duration;

/// Status of a subtask execution.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub enum SubtaskStatus {
    Pending,
    Running,
    Completed,
    Failed(String),
    Retrying { attempt: u32 },
}

/// Result of executing a single subtask.
#[derive(Debug, Clone)]
pub struct SubtaskResult {
    pub subtask_id: SubtaskId,
    pub agent_role: String,
    pub status: SubtaskStatus,
    pub output: String,
    pub duration: Duration,
    pub llm_turns: u32,
}

/// Executes a single subtask using a Rig agent.
pub struct WorkerAgent {
    shared_memory: SharedMemory,
    api_key: String,
    timeout: Duration,
}

impl WorkerAgent {
    pub fn new(shared_memory: SharedMemory, api_key: String, timeout: Duration) -> Self {
        Self {
            shared_memory,
            api_key,
            timeout,
        }
    }

    /// Execute a subtask: read inputs from shared memory, run agent, write output.
    pub async fn execute(&self, node: &DagNode) -> crate::Result<SubtaskResult> {
        let start = std::time::Instant::now();

        // Gather inputs from shared memory
        let mut context = String::new();
        for key in &node.input_keys {
            if let Some(val) = self.shared_memory.get::<String>(key) {
                context.push_str(&format!("## Input: {}\n{}\n\n", key, val));
            }
        }

        let full_prompt = if context.is_empty() {
            node.prompt.clone()
        } else {
            format!("{}\n\n---\nContext from previous subtasks:\n{}", node.prompt, context)
        };

        // Build and run a Rig agent
        let client = rig::providers::anthropic::ClientBuilder::new(&self.api_key).build();
        let agent = client
            .agent("claude-sonnet-4-20250514")
            .preamble(&format!("You are a {} agent. Complete the assigned task.", node.agent_role))
            .max_turns(10)
            .build();

        let response = tokio::time::timeout(
            self.timeout,
            rig::completion::Prompt::prompt(&agent, &full_prompt),
        )
        .await
        .map_err(|_| crate::SwarmError::Timeout(self.timeout))?
        .map_err(|e| crate::SwarmError::WorkerFailed {
            subtask_id: node.subtask_id.to_string(),
            message: e.to_string(),
        })?;

        // Store output in shared memory
        self.shared_memory.set(&node.output_key, response.clone());

        Ok(SubtaskResult {
            subtask_id: node.subtask_id.clone(),
            agent_role: node.agent_role.clone(),
            status: SubtaskStatus::Completed,
            output: response,
            duration: start.elapsed(),
            llm_turns: 0, // Rig doesn't expose turn count directly
        })
    }
}
```

```rust
    pub swarm_id: SwarmId,
    pub final_output: String,
    pub subtask_results: HashMap<String, SubtaskResult>,
    pub total_duration: Duration,
    pub total_llm_turns: u32,
}

/// Orchestrates plan -> execute waves -> merge cycle.
pub struct SwarmCoordinator {
    pub config: SwarmConfig,
    api_key: String,
}

impl SwarmCoordinator {
    pub fn new(config: SwarmConfig, api_key: String) -> Self {
        Self { config, api_key }
    }

    /// Main entry point: decompose, execute waves, merge.
    pub async fn run(
        &self,
        task_prompt: &str,
        progress_tx: Option<broadcast::Sender<String>>,
    ) -> Result<SwarmResult> {
        let swarm_id = SwarmId::new();
        let start = Instant::now();

        self.emit(&progress_tx, format!("[swarm:{}] Planning task decomposition...", swarm_id));

        // Phase 1: Plan
        let planner = SwarmPlanner::new(
            self.config.planner_model.clone(),
            self.api_key.clone(),
        );
        let dag = planner.plan(task_prompt).await?;
        let waves = dag.topological_waves();

        self.emit(&progress_tx, format!(
            "[swarm:{}] Plan: {} subtasks in {} waves",
            swarm_id, dag.node_count(), waves.len()
        ));

        // Phase 2: Execute waves
        let shared_memory = SharedMemory::new();
        let mut all_results: HashMap<String, SubtaskResult> = HashMap::new();

        for (wave_idx, wave) in waves.iter().enumerate() {
            self.emit(&progress_tx, format!(
                "[swarm:{}] Executing wave {}/{} ({} subtasks)",
                swarm_id, wave_idx + 1, waves.len(), wave.len()
            ));

            let mut handles = Vec::new();
            let semaphore = std::sync::Arc::new(tokio::sync::Semaphore::new(
                self.config.max_concurrent_workers,
            ));

            for node in wave {
                let worker = WorkerAgent::new(
                    shared_memory.clone(),
                    self.api_key.clone(),
                    self.config.worker_timeout,
                );
                let node_owned = (*node).clone();
                let sem = semaphore.clone();
                let ptx = progress_tx.clone();
                let sid = swarm_id.clone();

                let handle = tokio::spawn(async move {
                    let _permit = sem.acquire().await.unwrap();
                    if let Some(ref tx) = ptx {
                        let _ = tx.send(format!(
                            "[worker:{}:{}] Starting...",
                            node_owned.agent_role, node_owned.subtask_id
                        ));
                    }
                    worker.execute(&node_owned).await
                });
                handles.push(handle);
            }

            // Collect results from this wave
            for handle in handles {
                match handle.await {
                    Ok(Ok(result)) => {
                        all_results.insert(result.subtask_id.to_string(), result);
                    }
                    Ok(Err(e)) => {
                        warn!("Worker failed: {}", e);
                        return Err(e);
                    }
                    Err(e) => {
                        error!("Worker panicked: {}", e);
                        return Err(SwarmError::WorkerFailed {
                            subtask_id: "unknown".into(),
                            message: format!("Task panicked: {}", e),
                        });
                    }
                }
            }
        }

        // Phase 3: Merge
        let merger = OutputMerger::new(MergeStrategy::Concatenate);
        let outputs: Vec<(String, String)> = all_results
            .values()
            .map(|r| (r.agent_role.clone(), r.output.clone()))
            .collect();
        let final_output = merger.merge(&outputs);

        let total_turns: u32 = all_results.values().map(|r| r.llm_turns).sum();

        self.emit(&progress_tx, format!(
            "[swarm:{}] Complete: {}/{} subtasks, {:.1}s total",
            swarm_id,
            all_results.len(),
            dag.node_count(),
            start.elapsed().as_secs_f64()
        ));

        Ok(SwarmResult {
            swarm_id,
            final_output,
            subtask_results: all_results,
            total_duration: start.elapsed(),
            total_llm_turns: total_turns,
        })
    }

    fn emit(&self, tx: &Option<broadcast::Sender<String>>, msg: String) {
        info!("{}", msg);
        if let Some(tx) = tx {
            let _ = tx.send(msg);
        }
    }
}
```

```rust
// crates/mcclawd-channels/src/registry.rs

use crate::{Channel, ChannelKind, Envelope, SessionKey, OutboundChunk};
use std::collections::HashMap;
use std::sync::Arc;
use std::sync::atomic::AtomicU64;
use tokio::sync::{mpsc, RwLock};
use tokio_util::sync::CancellationToken;
use chrono::{DateTime, Utc};

#[derive(Debug, Clone)]
pub enum ChannelStatus {
    Starting,
    Running,
    Degraded(String),
    Reconnecting,
    Stopped,
    Failed(String),
}

struct ChannelEntry {
    channel: Arc<dyn Channel>,
    status: ChannelStatus,
    started_at: DateTime<Utc>,
    message_count: Arc<AtomicU64>,
}

pub struct ChannelRegistry {
    channels: RwLock<HashMap<ChannelKind, ChannelEntry>>,
    inbound_tx: mpsc::Sender<Envelope>,
    shutdown: CancellationToken,
}

impl ChannelRegistry {
    pub fn new(inbound_tx: mpsc::Sender<Envelope>, shutdown: CancellationToken) -> Self {
        Self {
            channels: RwLock::new(HashMap::new()),
            inbound_tx,
            shutdown,
        }
    }

    /// Register and start a new channel.
    pub async fn register(&self, channel: Arc<dyn Channel>) -> anyhow::Result<()> {
        let kind = channel.kind();
        let entry = ChannelEntry {
            channel: channel.clone(),
            status: ChannelStatus::Starting,
            started_at: Utc::now(),
            message_count: Arc::new(AtomicU64::new(0)),
        };

        let mut channels = self.channels.write().await;
        channels.insert(kind.clone(), entry);

        // Start the channel
        let tx = self.inbound_tx.clone();
        let shutdown = self.shutdown.clone();
        channel.start(tx, shutdown).await
            .map_err(|e| anyhow::anyhow!("Failed to start channel {:?}: {}", kind, e))?;

        if let Some(entry) = channels.get_mut(&kind) {
            entry.status = ChannelStatus::Running;
        }

        Ok(())
    }

    /// Unregister a channel.
    pub async fn unregister(&self, kind: &ChannelKind) -> anyhow::Result<()> {
        let mut channels = self.channels.write().await;
        channels.remove(kind);
        Ok(())
    }

    /// Route an outbound chunk to the correct channel.
    pub async fn send(
        &self,
        session_key: &SessionKey,
        chunk: OutboundChunk,
    ) -> anyhow::Result<()> {
        let kind = Self::kind_from_session(session_key);
        let channels = self.channels.read().await;
        let entry = channels.get(&kind)
            .ok_or_else(|| anyhow::anyhow!("No channel registered for {:?}", kind))?;
        entry.channel.send_chunk(session_key, chunk).await
            .map_err(|e| anyhow::anyhow!("Send failed: {}", e))
    }

    /// List active channel kinds.
    pub async fn active_channels(&self) -> Vec<ChannelKind> {
        self.channels.read().await.keys().cloned().collect()
    }

    /// Get statuses of all channels.
    pub async fn channel_statuses(&self) -> HashMap<ChannelKind, ChannelStatus> {
        self.channels.read().await
            .iter()
            .map(|(k, v)| (k.clone(), v.status.clone()))
            .collect()
    }

    fn kind_from_session(session_key: &SessionKey) -> ChannelKind {
        match session_key.channel.as_str() {
            "telegram" => ChannelKind::Telegram,
            "discord" => ChannelKind::Discord,
            "slack" => ChannelKind::Slack,
            "web" => ChannelKind::Web,
            _ => ChannelKind::Cli,
        }
    }
}
```

```rust
use mcclawd_channels::{
    Channel, ChannelCapabilities, ChannelKind, Envelope, MessageContent,
    OutboundChunk, SessionKey, StreamingMode,
};
use async_trait::async_trait;
use teloxide::prelude::*;

use tokio::sync::{mpsc, RwLock};

pub struct TelegramChannel {
    config: TelegramConfig,
    bot: Bot,
    /// Message ID of the "in-progress" message being edited (for streaming)
    progress_messages: RwLock<HashMap<String, i32>>,
}

impl TelegramChannel {
    pub fn new(config: TelegramConfig) -> Self {
        let bot = Bot::new(&config.token);
        Self {
            config,
            bot,
            progress_messages: RwLock::new(HashMap::new()),
        }
    }

    fn kind(&self) -> ChannelKind {
        ChannelKind::Telegram
    }

    fn display_name(&self) -> &str {
        "Telegram"
    }

    fn capabilities(&self) -> ChannelCapabilities {
        ChannelCapabilities {
            text: true,
            markdown: true,
            code_blocks: true,
            file_upload: true,
            max_file_size: Some(50 * 1024 * 1024), // 50MB Telegram limit
            supported_media_types: vec![
                "image/jpeg".into(), "image/png".into(),
                "application/pdf".into(), "text/plain".into(),
            ],
            message_edit: true,
            streaming_mode: StreamingMode::EditInPlace,
            max_message_length: Some(self.config.max_message_length),
        }
    }
}

/// Split a long message into chunks that fit within the limit.
pub fn split_message(text: &str, max_len: usize) -> Vec<String> {
    if text.len() <= max_len {
        return vec![text.to_string()];
    }

    let mut parts = Vec::new();
    let mut remaining = text;
    while !remaining.is_empty() {
        let end = std::cmp::min(remaining.len(), max_len);
        // Try to split at a newline boundary
        let split_at = remaining[..end]
            .rfind('\n')
            .map(|i| i + 1)
            .unwrap_or(end);
        parts.push(remaining[..split_at].to_string());
        remaining = &remaining[split_at..];
    }
    parts
}
```

```rust
// crates/mcclawd-channel-telegram/src/normalize.rs

use mcclawd_channels::{Envelope, SessionKey, MessageContent};

/// Normalize a plain text Telegram message into an Envelope.
pub fn normalize_text(chat_id: i64, user_id: i64, text: &str) -> Envelope {
    Envelope::text(
        SessionKey::new("telegram", &chat_id.to_string(), &user_id.to_string()),
        text,
    )
}

/// Normalize a command message into an Envelope.
pub fn normalize_command(chat_id: i64, user_id: i64, command: &str, args: Vec<String>) -> Envelope {
    let name = command.strip_prefix('/').unwrap_or(command).to_string();
    Envelope {
        id: uuid::Uuid::new_v4().to_string(),
        session_key: SessionKey::new("telegram", &chat_id.to_string(), &user_id.to_string()),
        content: MessageContent::Command { name, args },
        timestamp: chrono::Utc::now(),
        platform_message_id: None,
    }
}

/// Strip @bot_username mention from the beginning of a message.
pub fn strip_bot_mention(text: &str, bot_username: &str) -> String {
    let mention = format!("@{}", bot_username);
    text.strip_prefix(&mention)
        .map(|s| s.trim_start().to_string())
        .unwrap_or_else(|| text.to_string())
}
```

```rust
/// Test the full DAG -> wave -> merge pipeline without LLM calls.
#[test]
fn dag_to_merge_pipeline() {
    // Build a simple DAG: A -> B (research -> write)
    let mut dag = TaskDag::new();
    let a = dag.add_node(DagNode {
        subtask_id: SubtaskId::from("research".into()),
        agent_role: "researcher".into(),
        prompt: "research topic".into(),
        input_keys: vec![],
        output_key: "research_data".into(),
    });
    let b = dag.add_node(DagNode {
        subtask_id: SubtaskId::from("write".into()),
        agent_role: "writer".into(),
        prompt: "write report".into(),
        input_keys: vec!["research_data".into()],
        output_key: "final_report".into(),
    });
    dag.add_edge(a, b).unwrap();

    // Verify waves
    let waves = dag.topological_waves();
    assert_eq!(waves.len(), 2);
    assert_eq!(waves[0][0].agent_role, "researcher");
    assert_eq!(waves[1][0].agent_role, "writer");

    // Simulate shared memory flow
    let mem = SharedMemory::new();
    mem.set("research_data", "Found important data about X.".to_string());

    let input: Option<String> = mem.get("research_data");
    assert!(input.is_some());

    mem.set("final_report", "Report: Based on data about X, we conclude Y.".to_string());

    // Merge outputs
    let merger = OutputMerger::new(MergeStrategy::LastWriterWins {
        final_key: "final_report".into(),
    });
    let outputs = vec![
        ("research_data".to_string(), "Found important data about X.".to_string()),
        ("final_report".to_string(), "Report: Based on data about X, we conclude Y.".to_string()),
    ];
    let result = merger.merge(&outputs);
    assert!(result.contains("conclude Y"));
}
```
